# submit.txt -- HTCondor submit description for the power simulations
#
# Code for "Exact, Nonparametric Sensitivity Analysis for Observational Studies
# of Contingency Tables"
# Authors: Elaine K. Chiu and Hyunseung Kang
#
# Run from the top-level folder (the one with README.txt):
#   mkdir -p power_simulation/logs
#   condor_submit power_simulation/submit.txt
# One job is queued per (seed, gamma) row of power_simulation/params.txt. Each
# job runs power_analysis_weighted_four_gen_$(test).R with the arguments
# (seed, gamma, replications) and prints its number of rejections to the .out
# file in power_simulation/logs/; power_simulation/summarize_power.sh then
# sums the .out files of one test into summary_<test>.txt.
#
# Before submitting:
#   (i)  set container_image to your own Apptainer image; its contents are
#        described in README.txt, HOW TO REPRODUCE THE RESULTS, item 3;
#   (ii) set test to one of oracle, binary_outcome_1_2, binary_outcome_2_3,
#        Fisher_outcome_1_2, Fisher_outcome_2_3.

test            = oracle
container_image = /path/to/your/container.sif

executable = power_simulation/power_analysis_weighted_four_gen_$(test).R
arguments  = $(seed) $(gamma) 100

# library/ is transferred as a directory (no trailing slash), so that
# library/sensitivityIxJ.R and library/sensitivityIxJ.cpp resolve from the
# job's working directory exactly as they do on a single machine.
should_transfer_files   = YES
when_to_transfer_output = ON_EXIT
transfer_input_files    = library

log    = power_simulation/logs/$(test)_gamma_$(gamma)_seed_$(seed).log
error  = power_simulation/logs/$(test)_gamma_$(gamma)_seed_$(seed).err
output = power_simulation/logs/$(test)_gamma_$(gamma)_seed_$(seed).out

request_cpus   = 4
request_memory = 20GB
request_disk   = 10GB

queue seed, gamma from power_simulation/params.txt
